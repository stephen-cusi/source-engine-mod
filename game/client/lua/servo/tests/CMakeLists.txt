cmake_minimum_required(VERSION 3.16)
project(servo_backend_test LANGUAGES CXX)
enable_testing()
add_library(mock_servo SHARED mock_servo.cpp)
add_executable(servo_backend_test backend_test.cpp ../servo_backend.cpp)
target_compile_features(mock_servo PRIVATE cxx_std_11)
target_compile_features(servo_backend_test PRIVATE cxx_std_11)
if(CMAKE_DL_LIBS)
    target_link_libraries(servo_backend_test PRIVATE ${CMAKE_DL_LIBS})
endif()
find_package(Threads REQUIRED)
target_link_libraries(servo_backend_test PRIVATE Threads::Threads)
add_test(NAME backend COMMAND servo_backend_test $<TARGET_FILE:mock_servo>)
